                        className="p-3 border border-stone-200 rounded-xl hover:border-emerald-400 cursor-pointer transition-all flex items-center justify-between hover:shadow-md bg-stone-50"
                      >
                        <div>
                          <div className="font-bold text-sm text-stone-800">{p.title}</div>
                          <div className="text-xs text-stone-500 mt-1">預設印字：{p.text}</div>
                        </div>
                        <div className="flex gap-1">
                          <span className="w-4 h-4 rounded-full border" style={{ backgroundColor: p.cap }}></span>
                          <span className="w-4 h-4 rounded-full border" style={{ backgroundColor: p.body }}></span>
                          <span className="w-4 h-4 rounded-full border" style={{ backgroundColor: p.strap }}></span>
                        </div>
                      </div>
                    ))}
                  </div>
                </div>
              )}

              {/* TAB 4: AI 創意工作室 */}
              {activeTab === 'ai' && (
                <div className="space-y-4 bg-gradient-to-br from-violet-50 to-fuchsia-50 p-4 rounded-2xl border border-violet-100">
                  <div>
                    <label className="block text-xs font-bold text-violet-900 mb-1">輸入 AI 提示詞 (English Prompt)</label>
                    <input 
                      type="text"
                      value={aiPrompt}
                      onChange={(e) => setAiPrompt(e.target.value)}
                      placeholder="例如: Cute Shiba Inu, sakura blossom"
                      className="w-full px-3 py-2 border border-violet-200 rounded-lg text-sm focus:outline-none focus:ring-2 focus:ring-violet-400"
                    />
                  </div>

                  <div>
                    <label className="block text-xs font-bold text-violet-900 mb-1">選擇繪畫風格</label>
                    <select 
                      value={aiStyle}
                      onChange={(e) => setAiStyle(e.target.value)}
                      className="w-full px-3 py-2 border border-violet-200 rounded-lg text-sm bg-white focus:outline-none focus:ring-2 focus:ring-violet-400"
                    >
                      <option value="Vector Art Illustration">向量插畫 (Vector Art)</option>
                      <option value="Japanese Anime Style">日系動漫 (Anime Style)</option>
                      <option value="Watercolor Painting">水墨渲染 (Watercolor)</option>
                      <option value="3D Cute Render">3D 立體可愛 (3D Render)</option>
                      <option value="Photorealistic Studio">真實攝影 (Photorealistic)</option>
                    </select>
                  </div>

                  <button
                    onClick={handleAIGenerate}
                    disabled={isGenerating || !aiPrompt}
                    className="w-full py-2.5 bg-violet-600 hover:bg-violet-700 disabled:opacity-30 text-white font-bold text-sm rounded-xl transition-colors shadow-md flex items-center justify-center gap-2"
                  >
                    {isGenerating ? (
                      <>
                        <span className="animate-spin inline-block w-4 h-4 border-2 border-white border-t-transparent rounded-full"></span>
                        AI 生成中...
                      </>
                    ) : (
                      '✨ 生成專屬瓶身圖案'
                    )}
                  </button>
                  <p className="text-[10px] text-violet-600 text-center">Powered by Pollinations AI (Flux)</p>
                </div>
              )}

            </div>
          </main>

          {/* QR Code 分享彈窗 Modal */}
          {showQR && (
            <div className="fixed inset-0 bg-black/50 backdrop-blur-sm flex items-center justify-center z-50 p-4">
              <div className="bg-white rounded-3xl p-6 max-w-sm w-full text-center space-y-4 shadow-2xl">
                <h3 className="text-lg font-bold text-stone-800">分享您的客製矽水瓶</h3>
                <p className="text-xs text-stone-500">掃描下方 QR Code 即可傳送專屬配色體驗！</p>
                
                <div className="flex justify-center p-4 bg-stone-50 rounded-2xl">
                  <img 
                    src={`https://api.qrserver.com/v1/create-qr-code/?size=180x180&data=${encodeURIComponent(window.location.href)}`} 
                    alt="Design QR Code"
                    className="w-44 h-44 object-contain"
                  />
                </div>

                <button 
                  onClick={() => setShowQR(false)}
                  className="w-full py-2 bg-stone-200 hover:bg-stone-300 text-stone-800 rounded-xl text-sm font-semibold transition-colors"
                >
                  關閉
                </button>
              </div>
            </div>
          )}
        </div>
      );
    }

    // 將應用程式渲染至 DOM
    const root = ReactDOM.createRoot(document.getElementById('root'));
    root.render(<App />);
  </script>
